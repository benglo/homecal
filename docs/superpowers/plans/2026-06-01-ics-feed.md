# iCal Feed Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a read-only `GET /api/feed.ics` endpoint that serves all events (with native RRULE + EXDATE) and dinners as a subscribable iCalendar feed.

**Architecture:** Single new route file queries all masters + exceptions + dinners + categories from SQLite, builds a VCALENDAR via `ical-generator`, and streams the text/calendar response. Per-event try/catch so one bad record never breaks the whole feed.

**Tech Stack:** `ical-generator` (CJS-compatible via `dist/index.cjs`), node:test for testing, existing Fastify + better-sqlite3 backend.

### Task 4: Implement the feed builder and route

**Files:**

- [ ] **Step 1: Create the feed route**

```typescript
import type { FastifyInstance } from 'fastify';
import ical, { ICalCalendarMethod } from 'ical-generator';
import type { Category, EventMaster, Dinner } from '../model/types';
import type { CancelledEx } from '../repos/events';
import { listCategories } from '../repos/categories';
import { listAllMasters, listAllCancelledExceptions } from '../repos/events';
import { listAllDinners } from '../repos/dinners';

function isoToIcalDate(iso: string): string {
  return iso.slice(0, 10).replace(/-/g, '');
}

function isoToIcalDatetime(iso: string): string {
  return iso.replace(/[-:]/g, '').replace(/\.\d{3}/, '');
}

export function buildFeed(
  categories: Category[],
  masters: EventMaster[],
  exceptions: CancelledEx[],
  dinners: Dinner[]
): string {
  const cal = ical({
    name: 'Family Calendar',
    prodId: '-//homecal//EN',
    method: ICalCalendarMethod.PUBLISH,
    scale: 'GREGORIAN',
    x: { 'X-WR-TIMEZONE': 'Australia/Brisbane' },
  });

  const catMap = new Map(categories.map((c) => [c.id, c.name]));
  const exMap = new Map<string, CancelledEx[]>();
  for (const ex of exceptions) {
    const list = exMap.get(ex.eventId) ?? [];
    list.push(ex);
    exMap.set(ex.eventId, list);
  }

  for (const master of masters) {
    try {
      const event = cal.createEvent({
        id: `${master.id}@homecal`,
        start: new Date(master.start),
        end: new Date(master.end),
        allDay: master.allDay,
        summary: master.title,
        lastModified: new Date(master.updatedAt),
      });

      if (master.location) event.location(master.location);

      const catName = catMap.get(master.categoryId);
      if (catName) event.createCategory({ name: catName });

      if (master.rrule) {
        event.repeating(master.rrule);
      }

      const cancelled = exMap.get(master.id);
      if (cancelled?.length) {
        const excludeDates = cancelled.map((ex) => new Date(ex.occurrenceDate));
        event.repeating({
          ...(typeof event.repeating() === 'string'
            ? { freq: 'DAILY' as any }
            : (event.repeating() as any)),
          exclude: excludeDates,
        });
      }
    } catch {
      // Skip this event — one bad record must never break the whole feed.
    }
  }

  for (const dinner of dinners) {
    try {
      const nextDay = new Date(dinner.date + 'T00:00:00Z');
      nextDay.setUTCDate(nextDay.getUTCDate() + 1);
      const nextDayStr = nextDay.toISOString().slice(0, 10);

      cal.createEvent({
        id: `dinner-${dinner.date}@homecal`,
        start: new Date(dinner.date + 'T00:00:00Z'),
        end: new Date(nextDayStr + 'T00:00:00Z'),
        allDay: true,
        summary: `Dinner: ${dinner.meal}`,
        lastModified: new Date(dinner.updatedAt),
        categories: [{ name: 'Dinner' }],
      });
    } catch {
      // Skip bad dinner record.
    }
  }

  return cal.toString();
}

export async function feedRoutes(app: FastifyInstance): Promise<void> {
  app.get('/api/feed.ics', async (_req, reply) => {
    const categories = listCategories();
    const masters = listAllMasters();
    const exceptions = listAllCancelledExceptions();
    const dinners = listAllDinners();

    const ics = buildFeed(categories, masters, exceptions, dinners);

    reply
      .header('Content-Type', 'text/calendar; charset=utf-8')
      .header('Content-Disposition', 'inline; filename="homecal.ics"')
      .header('Cache-Control', 'no-cache')
      .send(ics);
  });
}
```
